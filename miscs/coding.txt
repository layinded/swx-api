@router.post("/{id}/revoke", response_model=ApiKeysRead)
def revoke_api_key(
    id: uuid.UUID,
    db: SessionDep
):
    api_key = db.get(ApiKeys, id)
    if not api_key:
        raise HTTPException(status_code=404, detail="API Key not found")

    api_key.revoked = True
    db.add(api_key)
    db.commit()
    db.refresh(api_key)
    return api_key
@router.post("/{id}/unrevoke", response_model=ApiKeysRead)

from datetime import datetime, timezone

class ApiKeysRepository:
    @staticmethod
    def filter_api_keys(
        db: SessionDep,
        revoked: Optional[bool] = None,
        expired: Optional[bool] = None,
        skip: int = 0,
        limit: int = 100
    ) -> List[ApiKeys]:
        query = select(ApiKeys)

        if revoked is not None:
            query = query.where(ApiKeys.revoked == revoked)
        if expired is not None:
            now = datetime.now(timezone.utc)
            if expired:
                query = query.where(ApiKeys.expires_at < now)
            else:
                query = query.where(ApiKeys.expires_at >= now)

        query = query.offset(skip).limit(limit)
        return db.exec(query).all()
@router.get("/filter", response_model=List[ApiKeysRead])
def filter_api_keys(
    revoked: Optional[bool] = Query(None),
    expired: Optional[bool] = Query(None),
    skip: int = 0,
    limit: int = 100,
    db: SessionDep = Depends(get_db)
):
    return ApiKeysRepository.filter_api_keys(
        db, revoked=revoked, expired=expired, skip=skip, limit=limit
    )
